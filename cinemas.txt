--This database is a little movie theatre focused solely on HOYTS Cinema and moviegoers who enjoy going to their favourite theatre. 
--You will be able to explore the different HOYTS Cinemas locations, learn about the most recent films that the theatre is showing,

--includes information and details on customer, order, seat, ticket, and item.

DROP view CinemaMovies CASCADE;

DROP table Cinema CASCADE;
DROP table CinemaToMovie CASCADE;
DROP table Movie CASCADE;
DROP table Ticket CASCADE;
DROP table Seat CASCADE;
DROP table Customer CASCADE;
DROP table Orders CASCADE;
DROP table OrderToItem CASCADE;
DROP table Item CASCADE;

CREATE table Movie
(
     movieName          varchar(50),
     movieYear          integer,
     durationMin        integer,
     rating             char(10),
     ratingDescription  text,

     CONSTRAINT MoviePK PRIMARY KEY (movieName, movieYear),
     CONSTRAINT check_rating CHECK (rating IN ('M', 'MA', 'PG')),
     CONSTRAINT check_durationMin CHECK (durationMin <=180)
);

CREATE table Cinema
(
     cinemaLocation      varchar(20), 
     starRating          decimal(2,1),
     
     CONSTRAINT CinemaPK PRIMARY KEY (cinemaLocation),
     CONSTRAINT check_starRating CHECK (starRating <= 5.0)
);

CREATE table CinemaToMovie 
(
     cinemaLocation     varchar(20),
     movieName          varchar(50),
     movieYear          integer,

     CONSTRAINT CinemaToMoviePK PRIMARY KEY (cinemaLocation, movieName, movieYear),
     CONSTRAINT CinemaToMovieFK_Cinema FOREIGN KEY (cinemaLocation) REFERENCES Cinema ON DELETE CASCADE,
     CONSTRAINT CinemaToMovieFK_Movie FOREIGN KEY (movieName, movieYear) REFERENCES Movie ON DELETE CASCADE
);

CREATE table Seat
(
     seatID           char(4),
     cinemaNo         char(2),
     seatType         varchar(20),
     seatDescription  text,

     CONSTRAINT SeatPK PRIMARY KEY (seatID, cinemaNo),
     CONSTRAINT check_seatID CHECK (seatID LIKE '_-_%')
);

CREATE table Customer
(
     customerID       integer,
     fName            varchar(10),
     surname          varchar(10),
     email            varchar(30),
     isMember         char(2),

     CONSTRAINT CustomerPK PRIMARY KEY (customerID),
     CONSTRAINT check_isMember CHECK (isMember IN ('Y', 'N'))
);

CREATE table Ticket 
(
     customerID      integer,
     movieName       varchar(50),
     movieYear       integer,
     cinemaLocation  varchar(20),
     seatID          char(4),
     cinemaNo        char(2),
     ticketID        integer,
     ticketType      varchar(10),
     ticketPrice     decimal(4,2),
     date            date,
     time            time,
    
     CONSTRAINT TicketPK PRIMARY KEY (ticketID),
     CONSTRAINT TicketFK_Seat FOREIGN KEY (seatID, cinemaNo) REFERENCES Seat ON DELETE RESTRICT,
     CONSTRAINT TicketFK_Customer FOREIGN KEY (customerID) REFERENCES Customer ON UPDATE CASCADE ON DELETE CASCADE,
     CONSTRAINT TicketFK_Movie FOREIGN KEY (movieName, movieYear) REFERENCES Movie ON DELETE CASCADE,
     CONSTRAINT TicketFK_Cinema FOREIGN KEY (cinemaLocation) REFERENCES Cinema ON UPDATE CASCADE ON DELETE CASCADE,
     CONSTRAINT check_ticketPrice CHECK (ticketPrice > 0)
);

CREATE table Orders
(
     customerID      integer,
     cinemaLocation  varchar(20),
     orderID         char(9),
     orderDate       date,

     CONSTRAINT OrdersPK PRIMARY KEY (orderID),
     CONSTRAINT OrdersFK_Customer FOREIGN KEY (customerID) REFERENCES Customer ON DELETE RESTRICT,
     CONSTRAINT OrdersFK_Cinema FOREIGN KEY (cinemaLocation) REFERENCES Cinema ON DELETE RESTRICT
);

CREATE table Item
(
     itemID          integer,
     itemName        varchar(20),
     price           decimal(4,2),

     CONSTRAINT ItemPK PRIMARY KEY (itemID),
     CONSTRAINT check_price CHECK (price >0)
);

CREATE table OrderToItem
(
     orderID         char(9),
     itemID          integer,
     quantity        integer,

     CONSTRAINT OrderToItemPK PRIMARY KEY (orderID, itemID),
     CONSTRAINT OrderToItemFK_Orders FOREIGN KEY (orderID) REFERENCES Orders ON DELETE CASCADE,
     CONSTRAINT OrderToItemFK_Item FOREIGN KEY (itemID) REFERENCES Item ON DELETE CASCADE,
     CONSTRAINT check_quantity CHECK (quantity >0)
);

CREATE view CinemaMovies AS
SELECT cinemaLocation as "Cinema", movieName as "Movie", starRating as "Rating", movieYear as "Release Year" 
FROM Cinema NATURAL JOIN CinemaToMovie NATURAL JOIN Movie 
WHERE starrating > 4.0;
